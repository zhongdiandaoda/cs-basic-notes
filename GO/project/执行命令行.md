# 执行命令行

```
import (
	"bytes"
	"fmt"
	"os/exec"
)

func main() {
	cmd := exec.Command("ls", "-l")
	var outBytes bytes.Buffer
	var errBytes bytes.Buffer
	cmd.Stdout = &outBytes
	cmd.Stderr = &errBytes
	err := cmd.Run()
	if err != nil {
		fmt.Println("err")
	}
	output := outBytes.String()
	fmt.Println(output)
}
```
